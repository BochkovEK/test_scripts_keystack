# Запуск миграции ВМ на указанный гипер
python ~/test_scripts_keystack/openstack-sdk/light_migration.py --source-host cdm-bl-pca05 --target-host cdm-bl-pca06

# Миграция CLI
source_host="" \
target_host="" \
echo "Migration all VMs from source_host to target_host" \
for vm_id in $(openstack server list --all-project --host $source_host --status ACTIVE -c ID -f value); do \
    echo "Migration VM $vm_id to $target_host..." \
    openstack server migrate --live-migration --host $target_host $vm_id; done

#или
src=""; tgt="" \
echo "Migrate VMs from $src to $tgt" \
openstack server list --all-project --host $src --status ACTIVE \
-c ID -f value | while read vm; do echo "Migrate $vm to $tgt"; openstack server migrate --live-migration --host $tgt $vm; done

watch -d openstack server migration list

openstack compute service set --up --enable cdm-bl-pca05 nova-compute
